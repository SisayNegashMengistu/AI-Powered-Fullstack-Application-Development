Quiz 1 not checked 
function job() {
    return new Promise(function(resolve, reject) {
        reject();
    });
}

let promise = job();

promise

.then(function() {
    console.log('Success 1');
})

.then(function() {
    console.log('Success 2');
})

.then(function() {
    console.log('Success 3');
})

.catch(function() {
    console.log('Error 1');
})

.then(function() {
    console.log('Success 4');
});

YOUR GUESS

ACTUAL OUTPUT

Why?

The promise job() returns rejects immediately, so the .then() calls for "Success 1/2/3" have no rejection handler — the rejection just skips past all three and falls straight through to .catch(), which logs Error 1. Because .catch() doesn't re-throw, it produces a new fulfilled promise, so the final .then() runs normally and logs Success 4.

Quiz 2 not checked 
function job(state) {
    return new Promise(function(resolve, reject) {
        if (state) {
            resolve('success');
        } else {
            reject('error');
        }
    });
}

let promise = job(true);

promise

.then(function(data) {
    console.log(data);

    return job(false);
})

.catch(function(error) {
    console.log(error);

    return 'Error caught';
})

.then(function(data) {
    console.log(data);

    return job(true);
})

.catch(function(error) {
    console.log(error);
});

YOUR GUESS

ACTUAL OUTPUT

Why?

job(true) resolves with 'success', so the first .then() logs it, then returns job(false), a rejecting promise. That rejection is caught by the following .catch(), which logs 'error' and returns the string 'Error caught' — a fulfilled value. The next .then() logs 'Error caught' and returns job(true), which resolves. Since that promise fulfills rather than rejects, the final .catch() has nothing to catch and never runs.

Quiz 3 not checked 
function job(state) {
    return new Promise(function(resolve, reject) {
        if (state) {
            resolve('success');
        } else {
            reject('error');
        }
    });
}

let promise = job(true);

promise

.then(function(data) {
    console.log(data);

    return job(true);
})

.then(function(data) {
    if (data !== 'victory') {
        throw 'Defeat';
    }

    return job(true);
})

.then(function(data) {
    console.log(data);
})

.catch(function(error) {
    console.log(error);

    return job(false);
})

.then(function(data) {
    console.log(data);

    return job(true);
})

.catch(function(error) {
    console.log(error);

    return 'Error caught';
})

.then(function(data) {
    console.log(data);

    return new Error('test');
})

.then(function(data) {
    console.log('Success:', data.message);
})

.catch(function(data) {
    console.log('Error:', data.message);
});

YOUR GUESS

ACTUAL OUTPUT

Why?

job(true) resolves 'success', logged by then #1, which returns another job(true). Then #2 receives 'success', which isn't 'victory', so it throws 'Defeat' — this rejects the chain, skipping then #3 entirely. The first .catch() logs 'Defeat' and returns job(false), a rejection, which skips the next .then() and lands on the second .catch(), logging 'error' and returning the fulfilled string 'Error caught'. The following .then() logs 'Error caught' and returns new Error('test') — an Error object is just a plain value here (not thrown), so the chain stays fulfilled, and the final .then() logs Success: test. Because nothing rejected after that, the last .catch() never fires.
